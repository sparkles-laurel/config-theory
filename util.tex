\newcommand{\definitionbox}[3]{%
  \begin{tcolorbox}[
      colback=defbg,
      colframe=defborder,
      coltitle=black,
      fonttitle=\bfseries,
      arc=5pt,
      boxrule=0.8pt,
      title={definition #1: #2}
  ]
  \label{#1}
  #3
  \end{tcolorbox}%
}
